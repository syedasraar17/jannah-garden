name: Build Android APK

on:
  workflow_dispatch:
  push:

jobs:
  apk:
    runs-on: ubuntu-latest
    container:
      image: barichello/godot-ci:4.3
    steps:
      - uses: actions/checkout@v4

      - name: Set up Godot export templates and Android settings
        run: |
          mkdir -v -p ~/.local/share/godot/export_templates/
          mkdir -v -p ~/.config/
          mv /root/.config/godot ~/.config/godot
          mv /root/.local/share/godot/export_templates/4.3.stable ~/.local/share/godot/export_templates/4.3.stable

      - name: Import project and export APK
        run: |
          mkdir -v -p build
          godot --headless --verbose --import || true
          godot --headless --verbose --export-debug "Android" build/JannahGarden.apk

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: JannahGarden-apk
          path: build/JannahGarden.apk
